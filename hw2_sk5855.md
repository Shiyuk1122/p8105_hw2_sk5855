``` r
subway_df = 
  readxl::read_xlsx("NYC_Transit_Subway_Entrance_And_Exit_Data.xlsx" , na = c("","NA",".")) |>
  janitor::clean_names()

subway_df_selected = 
  subway_df |>
  select(line:entry, vending, ada) |>
  mutate(
    entry = case_match(entry, "YES" ~ 1, "NO" ~ 0)
  )
```
